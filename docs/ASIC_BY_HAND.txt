# ASIC setup "by hand":

# Power up ASIC:

./pacman_util.py --write 0x24010 58000

./pacman_util.py --write 0x4000F010 0x10001

./pacman_util.py --write 0x40003B04 0x05281601
./pacman_util.py --write 0x40007B04 0x00001001
