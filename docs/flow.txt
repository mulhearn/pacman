# Basic work flows for pacman project:

# prevent accidentally sourcing the commands in this file:
return

# building from repo:
git clone https://github.com/mulhearn/pacman.git
# or, with ssh:
git clone git@github.com:mulhearn/pacman.git
cd pacman
source /tools/Xilinx/Vivado/2023.2/settings64.sh
# create the project from tcl, note that you specify target PACMAN HW version here (and only here):
vivado -mode batch -source vivado/tcl/create.tcl -tclargs hw1v5
# run systhesis, implementation, and write bit stream:
vivado -mode batch -source vivado/tcl/runsib.tcl

# connecting to UART
busybox microcom -s 115200 /dev/ttyUSB1
# (Use Ctrl-X to exit)

# run hello world demo:
xsct apps/hello/install.tcl
cd hello
xsct run.tcl

# run hw demostration bare-metal application
xsct apps/demo/install.tcl
cd demo
xsct run.tcl


# USAGE NOTE:
# With some versions of VIVADO, xsct tool under linux hangs occassionally at:
#
# $ xsct build.tcl
# Starting vitis. This could take few seconds...Eclipse:
# The Eclipse executable launcher no longer supports running with GTK + 2.x. Continuing using GTK+ 3.x.
# done
#
# In this case, you have to kill the hung processes yourself, e.g. like:
#
# (Ctrl-c)
pkill -9 -f eclipse

#building the FSBL: (You'll need this for petalinux packaging!)
xsct apps/pacman_fsbl/install.tcl
ls -alh products/pacman_fsbl.elf

#petalinux:
source /tools/Xilinx/PetaLinux/2023.2/tool/settings.sh
cd petalinux
./scripts/create_full.sh
cd full
../scripts/config.sh
../scripts/build.sh
../scripts/package.sh
cd images/linux
ls -alh BOOT.BIN image.ub boot.scr

# put those files on partition 1 (vfat) of an SD card,
# untar (as root) the rootfs.tar.gz into partion 2 (linux) of an SD card

# building just one user petalinux package (e.g. hwutil/pacman-server):
../scripts/build_hwutil.sh
# build tar file from the generated RPM file:
mkdir tmp
cd tmp/
rpm2cpio ../hwutil-latest.rpm | cpio -idmv
tar -czf ../hwutil-latest.tar.gz ./
cd ..
rm -fr tmp
